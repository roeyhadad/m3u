Option Explicit

Dim WithEvents objInspectors As Outlook.Inspectors
Dim g_AutoApprovalInProgress As Boolean

' רץ כש-Outlook עולה
Private Sub Application_Startup()
    Set objInspectors = Application.Inspectors
End Sub

' רץ על כל מייל חדש שנכנס
Private Sub Application_NewMailEx(ByVal EntryIDCollection As String)
    Dim itm As Object
    Dim mail As Outlook.MailItem

    On Error GoTo ExitPoint

    Set itm = Application.Session.GetItemFromID(EntryIDCollection)

    If TypeOf itm Is Outlook.MailItem Then
        Set mail = itm

        ' תנאי לזיהוי המייל הרלוונטי – תוכל לכוונן לפי Subject/Body
        ' למשל, אם תמיד יש "בקשה מספר CHG" בגוף:
        If InStr(mail.Body, "לאישור הבקשה") > 0 Then

            g_AutoApprovalInProgress = True   ' מסמן שאנחנו בתהליך אוטומטי

            ' פותח את המייל כדי שנוכל להגיע להייפרלינקים
            mail.Display

            ' לוחץ על הלינק "לאישור הבקשה"
            Call ClickApprovalLink(mail)
        End If
    End If

ExitPoint:
    Exit Sub
End Sub

' לוחץ על ההייפרלינק עם הטקסט "לאישור הבקשה"
Private Sub ClickApprovalLink(ByVal mail As Outlook.MailItem)
    Dim doc As Object
    Dim links As Object
    Dim h As Object

    On Error GoTo ExitPoint

    Set doc = mail.GetInspector.WordEditor
    Set links = doc.Hyperlinks

    For Each h In links
        ' אם טקסט ההייפרלינק הוא "לאישור הבקשה" (או חלק ממנו)
        If InStr(h.TextToDisplay, "אישור הבקשה") > 0 Then
            h.Follow    ' כמו לחיצה על הלינק
            Exit For
        End If
    Next h

ExitPoint:
    Exit Sub
End Sub

' רץ כל פעם שנפתח חלון חדש (Inspector חדש)
Private Sub objInspectors_NewInspector(ByVal Inspector As Outlook.Inspector)
    Dim itm As Object
    Dim mail As Outlook.MailItem

    On Error GoTo ExitPoint

    ' נטפל רק אם אנחנו בתהליך האוטומטי
    If g_AutoApprovalInProgress = False Then Exit Sub

    Set itm = Inspector.CurrentItem
    If TypeOf itm Is Outlook.MailItem Then
        Set mail = itm

        ' לוודא שזה באמת מייל האישור (הנושא כמו בצילום: Re:CHG... - approve)
        If InStr(mail.Subject, "- approve") > 0 Then
            ' שולח את המייל אוטומטית
            mail.Send
        End If
    End If

    ' מסיים את הסשן של האוטומציה
    g_AutoApprovalInProgress = False

ExitPoint:
    Exit Sub
End Sub
