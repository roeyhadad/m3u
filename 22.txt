Option Explicit

' Runs on every new mail that arrives
Private Sub Application_NewMailEx(ByVal EntryIDCollection As String)
    Dim arrIDs() As String
    Dim i As Long
    Dim itm As Object
    Dim mail As Outlook.MailItem
    Dim senderAddr As String

    arrIDs = Split(EntryIDCollection, ",")

    For i = LBound(arrIDs) To UBound(arrIDs)
        Set itm = Application.Session.GetItemFromID(arrIDs(i))
        If TypeOf itm Is Outlook.MailItem Then
            Set mail = itm

            ' Get sender SMTP address (works with Exchange as well)
            senderAddr = LCase(GetSenderSmtpAddress(mail))

            ' Filter:
            ' 1) subject contains "CHG"
            ' 2) mail is from specific sender
            If InStr(mail.Subject, "CHG") > 0 _
               And senderAddr = "change.system@bankleumi.co.il" Then   ' <<< CHANGE THIS

                AutoApprove mail
            End If
        End If
    Next i
End Sub

' Return sender SMTP address in a robust way
Private Function GetSenderSmtpAddress(ByVal mail As Outlook.MailItem) As String
    Dim addr As String
    Dim exch As Outlook.ExchangeUser

    On Error Resume Next

    If Not mail.Sender Is Nothing Then
        Set exch = mail.Sender.GetExchangeUser
        If Not exch Is Nothing Then
            addr = exch.PrimarySmtpAddress
        Else
            addr = mail.SenderEmailAddress
        End If
    Else
        addr = mail.SenderEmailAddress
    End If

    On Error GoTo 0
    GetSenderSmtpAddress = addr
End Function

' Read 2nd hyperlink, parse its mailto:, and send approval mail
Private Sub AutoApprove(ByVal mail As Outlook.MailItem)
    Dim doc As Object
    Dim links As Object
    Dim h As Object
    Dim addr As String

    On Error GoTo ExitPoint

    ' Need to display so WordEditor is available
    mail.Display

    Set doc = mail.GetInspector.WordEditor
    Set links = doc.Hyperlinks

    ' We expect at least 2 links: the 2nd is "Approve request"
    If links.Count < 2 Then GoTo ExitPoint

    Set h = links(2)
    addr = h.Address           ' Example: mailto:Auto_SN?subject=...&body=...

    If LCase(Left$(addr, 7)) <> "mailto:" Then GoTo ExitPoint

    SendFromMailto addr

ExitPoint:
End Sub

' Parse mailto: URL and send a mail with same To/Subject/Body
Private Sub SendFromMailto(ByVal mailtoUrl As String)
    Dim rest As String
    Dim toPart As String
    Dim query As String
    Dim pos As Long
    Dim kv() As String
    Dim pair As Variant
    Dim key As String, val As String
    Dim subj As String, body As String
    Dim m As Outlook.MailItem

    ' Strip "mailto:"
    rest = Mid$(mailtoUrl, 8)

    pos = InStr(rest, "?")
    If pos > 0 Then
        toPart = Left$(rest, pos - 1)
        query = Mid$(rest, pos + 1)
    Else
        toPart = rest
        query = ""
    End If

    If Len(query) > 0 Then
        kv = Split(query, "&")
        For Each pair In kv
            pos = InStr(CStr(pair), "=")
            If pos > 0 Then
                key = LCase(Left$(pair, pos - 1))
                val = Mid$(pair, pos + 1)
                val = UrlDecode(val)
                If key = "subject" Then subj = val
                If key = "body" Then body = val
            End If
        Next pair
    End If

    Set m = Application.CreateItem(olMailItem)
    With m
        .To = toPart
        If subj <> "" Then .Subject = subj
        If body <> "" Then .Body = body
        .Send
    End With
End Sub

' Simple URL-decoder for mailto parameters
Private Function UrlDecode(ByVal s As String) As String
    Dim i As Long
    Dim ch As String
    Dim hex As String
    Dim result As String

    i = 1
    While i <= Len(s)
        ch = Mid$(s, i, 1)
        If ch = "+" Then
            result = result & " "
            i = i + 1
        ElseIf ch = "%" And i + 2 <= Len(s) Then
            hex = Mid$(s, i + 1, 2)
            result = result & Chr$(CLng("&H" & hex))
            i = i + 3
        Else
            result = result & ch
            i = i + 1
        End If
    Wend

    UrlDecode = result
End Function
